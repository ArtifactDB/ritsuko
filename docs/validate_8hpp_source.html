81inline void validate_1d_pointers(const H5::DataSet& data, hsize_t full_length, hsize_t heap_length, const Validate1dPointersOptions& options) {
82 validate_pointer_datatype(data, std::numeric_limits<Offset_>::digits, std::numeric_limits<Length_>::digits);
83 assert(data.getSpace().getSimpleExtentNdims() == 1);
84
85 const auto& plist = data.getCreatePlist();
86 hsize_t block_size = 0;
87 if (plist.getLayout() == H5D_CHUNKED) {
88 plist.getChunk(1, &block_size);
89 } else {
90 block_size = std::min(full_length, options.contiguous_chunk_size);
91 }
92
93 H5::DataSpace mspace(1, &block_size), dspace(1, &full_length);
94 auto buffer = sanisizer::create<std::vector<Pointer<Offset_, Length_> > >(block_size);
95 auto dtype = define_pointer_datatype<Offset_, Length_>();
96
97 hsize_t i = 0;
98 while (i < full_length) {
99 const auto available = sanisizer::min(full_length - i, block_size);
100 constexpr hsize_t zero = 0;
101 mspace.selectHyperslab(H5S_SELECT_SET, &available, &zero);
102 dspace.selectHyperslab(H5S_SELECT_SET, &available, &i);
103
104 data.read(buffer.data(), dtype, mspace, dspace);
105 for (I<decltype(available)> j = 0; j < available; ++j) {
106 const auto& val = buffer[j];
107 if (is_Pointer_out_of_range(val, heap_length)) {
108 throw std::runtime_error("compressed VLS array pointers at '" + hdf5::get_name(data) + "' are out of range of the heap");
109 }
110 }
111
112 i += available;
113 }

143void validate_nd_pointers(const H5::DataSet& data, const std::vector<hsize_t>& dimensions, hsize_t heap_length, const ValidateNdPointersOptions& options) {
144 validate_pointer_datatype(data, std::numeric_limits<Offset_>::digits, std::numeric_limits<Length_>::digits);
145 assert(data.getSpace().getSimpleExtentNdims() > 0);
146
147 // Cast of 'ndim' to 'int' is implicitly safe if the assertion holds.
148 const auto ndim = dimensions.size();
149 assert(sanisizer::is_equal(ndim, data.getSpace().getSimpleExtentNdims()));
150
151 std::vector<hsize_t> chunk_dims;
152 const auto& plist = data.getCreatePlist();
153 if (plist.getLayout() == H5D_CHUNKED) {
154 chunk_dims.resize(ndim); // No need to check this, dimensions is of the same type as chunk_dims.
155 plist.getChunk(ndim, chunk_dims.data());
156 } else {
157 chunk_dims = hdf5::mock_contiguous_chunks(dimensions, options.contiguous_chunk_size);
158 }
159
160 hdf5::IterateChunks iter(dimensions, chunk_dims);
161 H5::DataSpace fspace(ndim, dimensions.data());
162 H5::DataSpace mspace(ndim, iter.chunk_dimensions().data());
163 auto buffer = sanisizer::create<std::vector<Pointer<Offset_, Length_> > >(mspace.getSimpleExtentNpoints());
164 auto dtype = define_pointer_datatype<Offset_, Length_>();
165
166 while (iter.advance()) {
167 const auto& curcount = iter.counts();
168 mspace.setExtentSimple(ndim, curcount.data());
169 fspace.selectHyperslab(H5S_SELECT_SET, curcount.data(), iter.starts().data());
170
171 data.read(buffer.data(), dtype, mspace, fspace);
172 const auto available = mspace.getSimpleExtentNpoints();
173 for (I<decltype(available)> i = 0; i < available; ++i) {
174 const auto& val = buffer[i];
175 if (is_Pointer_out_of_range(val, heap_length)) {
176 throw std::runtime_error("compressed VLS array pointers at '" + hdf5::get_name(data) + "' are out of range of the heap");
177 }
178 }
179 }

191inline hsize_t validate_heap(const H5::DataSet& data) {
192 if (data.getTypeClass() != H5T_INTEGER) {
193 throw std::runtime_error("expected an integer datatype for the compressed VLS heap at '" + hdf5::get_name(data) + "'");
194 }
195 if (hdf5::exceeds_integer_limit(data.getIntType(), 8, false)) {
196 throw std::runtime_error("expected 8-bit unsigned integers for the compressed VLS heap at '" + hdf5::get_name(data) + "'");
197 }
198 auto dspace = data.getSpace();
199 if (dspace.getSimpleExtentNdims() != 1) {
200 throw std::runtime_error("expected a 1-dimensional dataset for the compressed VLS heap at '" + hdf5::get_name(data) + "'");
201 }
202 hsize_t len;
203 dspace.getSimpleExtentDims(&len);
204 return len;
205}
