34 static_assert(std::is_integral<Length_>::value);
35
36 if (data.getTypeClass() != H5T_COMPOUND) {

90template<typename Offset_, typename Length_>
91inline void validate_1d_pointers(const H5::DataSet& data, hsize_t full_length, hsize_t heap_length) {
92 validate_pointers<Offset_, Length_>(data);
93
94 assert(data.getSpace().getSimpleExtentNdims() == 1);
95
96 const auto& plist = data.getCreatePlist();
97 hsize_t block_size = 0;
98 if (plist.getLayout() == H5D_CHUNKED) {
99 plist.getChunk(1, &block_size);
100 } else {
101 // Hard-coding the mock chunk size for non-chunked datasets,
102 // it's not worth requiring an extra function argument to customize this.
103 block_size = sanisizer::min(full_length, 10000);
104 }
105
106 H5::DataSpace mspace(1, &block_size), dspace(1, &full_length);
107 auto buffer = sanisizer::create<std::vector<Pointer<Offset_, Length_> > >(block_size);
108 auto dtype = define_pointer_datatype<Offset_, Length_>();
109
110 hsize_t i = 0;
111 while (i < full_length) {
112 const auto available = sanisizer::min(full_length - i, block_size);
113 constexpr hsize_t zero = 0;
114 mspace.selectHyperslab(H5S_SELECT_SET, &available, &zero);
115 dspace.selectHyperslab(H5S_SELECT_SET, &available, &i);
116
117 data.read(buffer.data(), dtype, mspace, dspace);
118 for (I<decltype(available)> j = 0; j < available; ++j) {
119 const auto& val = buffer[j];
120 if (is_pointer_out_of_range(val.offset, val.length, heap_length)) {
121 throw std::runtime_error("VLS array pointers at '" + hdf5::get_name(data) + "' are out of range of the heap");
122 }
123 }
124
125 i += available;
126 }
127}

143template<typename Offset_, typename Length_>
144void validate_nd_pointers(const H5::DataSet& data, const std::vector<hsize_t>& dimensions, hsize_t heap_length) {
145 validate_pointers<Offset_, Length_>(data);
146
147 assert(data.getSpace().getSimpleExtentNdims() > 0);
148
149 // Cast of 'ndim' to 'int' is implicitly safe if the assertion holds.
150 const auto ndim = dimensions.size();
151 assert(sanisizer::is_equal(ndim, data.getSpace().getSimpleExtentNdims()));
152
153 std::vector<hsize_t> chunk_dims;
154 const auto& plist = data.getCreatePlist();
155 if (plist.getLayout() == H5D_CHUNKED) {
156 chunk_dims.resize(ndim); // No need to check this, dimensions is of the same type as chunk_dims.
157 plist.getChunk(ndim, chunk_dims.data());
158 } else {
159 chunk_dims = hdf5::mock_contiguous_chunks(dimensions, 10000); // Hard-coding the upper bound to save ourselves an argument.
160 }
161
162 hdf5::IterateChunks iter(dimensions, chunk_dims);
163 H5::DataSpace fspace(ndim, dimensions.data());
164 H5::DataSpace mspace(ndim, iter.chunk_dimensions().data());
165 auto buffer = sanisizer::create<std::vector<Pointer<Offset_, Length_> > >(mspace.getSimpleExtentNpoints());
166 auto dtype = define_pointer_datatype<Offset_, Length_>();
167
168 while (iter.advance()) {
169 const auto& curcount = iter.counts();
170 mspace.setExtentSimple(ndim, curcount.data());
171 fspace.selectHyperslab(H5S_SELECT_SET, curcount.data(), iter.starts().data());
172
173 data.read(buffer.data(), dtype, mspace, fspace);
174 const auto available = mspace.getSimpleExtentNpoints();
175 for (I<decltype(available)> i = 0; i < available; ++i) {
176 const auto& val = buffer[i];
177 if (is_pointer_out_of_range(val.offset, val.length, heap_length)) {
178 throw std::runtime_error("VLS array pointers at '" + hdf5::get_name(data) + "' are out of range of the heap");
179 }
180 }
181 }

191inline void validate_heap(const H5::DataSet& data) {
192 if (data.getTypeClass() != H5T_INTEGER) {
193 throw std::runtime_error("expected an integer datatype for the compressed VLS heap at '" + hdf5::get_name(data) + "'");
194 }
195 if (hdf5::exceeds_integer_limit(data.getIntType(), 8, false)) {
196 throw std::runtime_error("expected 8-bit unsigned integers for the compressed VLS heap at '" + hdf5::get_name(data) + "'");
197 }
198 if (data.getSpace().getSimpleExtentNdims() != 1) {
199 throw std::runtime_error("expected a 1-dimensional dataset for the compressed VLS heap at '" + hdf5::get_name(data) + "'");
