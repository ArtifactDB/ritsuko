121 // Cast of 'ndims' to 'int' is implicitly safe if the assertion holds.

127 if (plist.getLayout() == H5D_CHUNKED) {
128 chunk_dims.resize(ndims); // this is safe as 'dimensions' and 'chunk_dims' have the same size_type.
129 plist.getChunk(dimensions.size(), chunk_dims.data());
130 } else {
131 // Hard-coding this to save ourselves an argument.
132 chunk_dims = mock_contiguous_chunks(dimensions, 10000);
133 }
134

136

141
142 while (iter.advance()) {
