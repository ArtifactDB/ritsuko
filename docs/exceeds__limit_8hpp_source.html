29inline bool exceeds_integer_limit(const H5::IntType& itype, std::size_t precision, bool is_signed) {
30 const auto sign = itype.getSign();
31 if (sign == H5T_SGN_NONE) {
32 if (is_signed) {
33 return (itype.getPrecision() >= precision); // equality, as one bit of the limiting type is used for the sign.
34 } else {
35 return (itype.getPrecision() > precision);
36 }
37 } else if (sign == H5T_SGN_2) {

41 return true;

46 return true;
