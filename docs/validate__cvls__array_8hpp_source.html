34inline void validate_1d_cvls_array(const H5::DataSet& handle, hsize_t full_length, hsize_t heap_length, hsize_t buffer_size) {
35 hsize_t block_size = hdf5::pick_1d_block_size(handle.getCreatePlist(), full_length, buffer_size);
36 H5::DataSpace mspace(1, &block_size), dspace(1, &full_length);
37 std::vector<CvlsPointer<Offset_, Length_> > buffer(block_size);
38 auto dtype = define_cvls_pointer_datatype<Offset_, Length_>();
39
40 for (hsize_t i = 0; i < full_length; i += block_size) {
41 auto available = std::min(full_length - i, block_size);
42 constexpr hsize_t zero = 0;
43 mspace.selectHyperslab(H5S_SELECT_SET, &available, &zero);
44 dspace.selectHyperslab(H5S_SELECT_SET, &available, &i);
45
46 handle.read(buffer.data(), dtype, mspace, dspace);
47 for (hsize_t j = 0; j < available; ++j) {
48 const auto& val = buffer[j];
49 hsize_t start = val.offset;
50 hsize_t count = val.length;
51 if (start > heap_length || start + count > heap_length) {
52 throw std::runtime_error("VLS array pointers at '" + hdf5::get_name(handle) + "' are out of range of the heap");
53 }
54 }
55 }

68void validate_nd_cvls_array(const H5::DataSet& handle, const std::vector<hsize_t>& dimensions, hsize_t heap_length, hsize_t buffer_size) {
69 std::vector<CvlsPointer<Offset_, Length_> > buffer;
70 auto dtype = define_cvls_pointer_datatype<Offset_, Length_>();
71 auto blocks = hdf5::pick_nd_block_dimensions(handle.getCreatePlist(), dimensions, buffer_size);
72 hdf5::IterateNdDataset iter(dimensions, blocks);
73
74 while (!iter.finished()) {
75 buffer.resize(iter.current_block_size());
76
77 // Scope this to ensure that 'mspace' doesn't get changed by
78 // 'iter.next()' before the destructor is called.
79 {
80 const auto& mspace = iter.memory_space();
81 handle.read(buffer.data(), dtype, mspace, iter.file_space());
82 for (const auto& val : buffer) {
83 hsize_t start = val.offset;
84 hsize_t count = val.length;
85 if (start > heap_length || start + count > heap_length) {
86 throw std::runtime_error("VLS array pointers at '" + hdf5::get_name(handle) + "' are out of range of the heap");
87 }
88 }
89 }
